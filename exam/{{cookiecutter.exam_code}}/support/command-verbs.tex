\newglossaryentry{determine}{name={determine},description={Obtain the only possible answer}}
\newglossaryentry{develop}{name={develop},description={Improve incrementally, elaborate or expand in detail. Evolve to a more advanced or effective state}}
\newglossaryentry{differentiate}{name={differentiate},description={Obtain the derivative of a function}}

\newglossaryentry{estimate}{name={estimate},description={Obtain an approximate value for an unknown quantity}}

\newglossaryentry{examine}{name={examine},description={Consider an argument or concept in a way that uncovers assumptions and interrelationships of the issue}}
\newglossaryentry{explain}{name={explain},description={Give a detailed account including reasons or causes}}
\newglossaryentry{explore}{name={explore},description={Undertake a systematic process of discovery}}
\newglossaryentry{find}{name={find},description={Obtain an answer showing relevant stages in the working}}
\newglossaryentry{formulate}{name={formulate},description={Express precisely and systematically the relevant concept(s) or argument(s)}}

\newglossaryentry{otherwise}{name={otherwise},description={It is suggested that the preceding work is used, but other methods could also receive credit}}

\newglossaryentry{integrate}{name={integrate},description={Obtain the integral of a function}}

\newglossaryentry{investigate}{name={investigate},description={Observe, study, or make a detailed and systematic examination in order to establish facts and reach new conclusions}}
\newglossaryentry{justify}{name={justify},description={Give valid reasons or evidence to support an answer or conclusion (see also \emph{Explain})}}
\newglossaryentry{label}{name={label},description={Add a title, labels or brief explanations to a diagram or graph}}

\newglossaryentry{outline}{name={outline},description={Give a brief account or summary}}

\newglossaryentry{prioritize}{name={prioritize},description={Give relative importance to, or put in an order of preference}}
\newglossaryentry{prove}{name={prove},description={Use a sequence of logical steps to obtain the required result in a formal way}}
\newglossaryentry{select}{name={select},description={Choose from a list or group}}
\newglossaryentry{show}{name={show},description={Give the steps in a calculation or derivation}}
\newglossaryentry{showthat}{name={show that},description={Obtain the required result (possibly using information given) without the formality of proof}}
\newglossaryentry{sketch}{name={sketch},description={Represent by means of a diagram or graph (labelled as appropriate). The sketch should give a general idea of the required shape or relationship, and should include relevant features}}
\newglossaryentry{solve}{name={solve},description={Obtain the answer(s) using algebraic, numerical or graphical methods}}
\newglossaryentry{state}{name={state},description={Give a specific name, value or other brief answer without explanation or calculation}}
\newglossaryentry{suggest}{name={suggest},description={Propose a solution, hypothesis or other possible answer}}
\newglossaryentry{summarize}{name={summarize},description={Abstract a general theme or major point(s)}}
\newglossaryentry{synthesize}{name={synthesize},description={Combine different ideas in order to create new understanding}}

\newglossaryentry{verify}{name={verify},description={Provide evidence that validates the result}}
